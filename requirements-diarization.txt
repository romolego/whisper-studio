# Зависимости отдельной среды для разметки спикеров (Pyannote и Nemotron).
#
# Отдельная среда, а не общая с сервисом, потому что у Pyannote свой torch:
# на машине с видеокартой это сборка с CUDA весом около 4.6 ГБ, и тащить её в
# среду распознавания незачем — Whisper считает через CTranslate2 и torch не
# использует вовсе.
#
# torch и torchaudio ставятся отдельной командой из индекса PyTorch, а не
# отсюда: на машине без видеокарты нужна процессорная сборка (около 250 МБ
# вместо 2.5 ГБ). Этим занимается scripts\install.ps1. К моменту установки
# этого файла torch уже стоит, и pip его не трогает.
pyannote.audio==4.0.7

# Звук читается soundfile, а не torchcodec: torchcodec тянется как зависимость
# Pyannote, но на Windows часто не находит библиотеки FFmpeg и падает при
# импорте. diarization_worker.py грузит WAV через soundfile и гасит
# предупреждение torchcodec, поэтому его поломка ни на что не влияет.
soundfile==0.14.0

# Второй движок разметки — NVIDIA Nemotron 3 Diarization (сентябрь 2026). Живёт
# в этой же среде: ему нужен тот же torch, что и Pyannote. Поддержку модели
# влили в transformers 23.09.2026, уже после выпуска 5.17.0, поэтому версия
# закреплена коммитом из исходников. Когда выйдет 5.18 — строку можно заменить
# на обычное transformers>=5.18. В офлайн-наборе колёс архива этого коммита
# нет, и установщик доберёт его из сети.
transformers @ https://github.com/huggingface/transformers/archive/c8b81b63232be35ab1774dd3cabbf499d8b9808f.zip

# Извлекатель признаков Nemotron строит мел-фильтры через librosa.
librosa==0.11.0

# Версия закреплена, хотя пакет и не используется. Своей зависимости от torch
# torchcodec не объявляет, поэтому pip свободен взять свежий torchcodec под
# старый torch — и собрать пару, которая падает при импорте. 0.15.0 — та, что
# стоит в рабочей среде рядом с torch 2.11.
torchcodec==0.15.0
